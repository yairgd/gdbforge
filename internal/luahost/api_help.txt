gdbforge.print(...)
  Append a line to :b io ([lua] prefix). In the Lua REPL, output stays in the lua pane.
  Example: gdbforge.print("hello")

gdbforge.help([topic])
  Show gdbforge API reference (:b io or REPL). Optional topic filters entries.
  Example: gdbforge.help("gdb")

gdbforge.register(name, fn)
  Expose fn as :lua name [args…].
  Example: gdbforge.register("probe", function() gdbforge.print("ok") end)

gdbforge.complete_args(fn)
  Tab-complete :lua <this-script> arguments. fn(token, index, prior…) → list of strings.
  index is 1-based; prior… are earlier args on the command line.
  Example: gdbforge.complete_args(function(t, i) return {"baremetal", "zephyr"} end)

gdbforge.gdb(cmd)
  Send one CLI line to the GDB/Delve console (same as typing in the gdb pane).
  Example: gdbforge.gdb("break main")

gdbforge.open_buffer(name)
  Focus a pane: "code", "gdb", "io", "lua", or a ModeLua buffer (e.g. "snake").
  Example: gdbforge.open_buffer("gdb")

gdbforge.program()
  Debuggee path from the gdbforge session ("" if none).
  Example: gdbforge.print(gdbforge.program())

gdbforge.debugger_path()
  Path of the active debugger binary (gdb or dlv).
  Example: gdbforge.print(gdbforge.debugger_path())

gdbforge.current_file() / gdbforge.current_line()
  Browse cursor file (absolute path) and 1-based line in Code.
  Example: gdbforge.print(gdbforge.current_file() .. ":" .. gdbforge.current_line())

gdbforge.stop_file() / gdbforge.stop_line()
  Stop PC (━━▶) source path and 1-based line.
  Example: gdbforge.print(gdbforge.stop_file() .. ":" .. gdbforge.stop_line())

gdbforge.run(argv…)
  Interactive shell / :! — replaces the focused pane with Exec.
  Example: gdbforge.run("make", "clean")

gdbforge.spawn(argv…)
  Background PTY process; keeps Code focused (J-Link, helpers, etc.).
  Example: gdbforge.spawn("JLinkGDBServer", "-port", "2331")

gdbforge.foreground(argv…)
  Suspend TUI, run on real tty, resume (vim, less, …).
  Example: gdbforge.foreground("vim", "+42", "main.c")

gdbforge.spawn_terminal(argv…)
  Open a real terminal emulator running argv (gdbserver TUI, dlv, …).
  Example: gdbforge.spawn_terminal("gdbserver", "localhost:2345")

gdbforge.open_external_tty()
  Hold a pts in kitty/xterm; returns path for inferior I/O.
  Example: local p = gdbforge.open_external_tty()

gdbforge.set_inferior_tty(path)
  GDB: -inferior-tty-set. Delve: restart with --tty. Use "internal" for gdbforge IO.
  Example: gdbforge.set_inferior_tty("internal")

gdbforge.dlv_connect(addr)
  Replace session with dlv connect (host:port or :port).
  Example: gdbforge.dlv_connect("127.0.0.1:2345")

gdbforge.spawn_dlv_headless(port [, args…])
  Headless dlv in a terminal; then use dlv_connect.
  Example: gdbforge.spawn_dlv_headless("2345")

gdbforge.wait_port(host_port [, timeout_sec])
  Block until TCP accepts (default ~10s, max 120s). Returns true/false.
  Example: gdbforge.wait_port("2345", 30)

gdbforge.sleep(seconds)
  Block the Lua VM (respects Ctrl-C job cancel).
  Example: gdbforge.sleep(0.5)

gdbforge.system(cmd)
  Run sh -c; returns exit status and combined stdout+stderr.
  Example: local st, out = gdbforge.system("uname -a")

gdbforge.lua_dir()
  Directory of the current script file (sidecars, XML, configs).
  Example: gdbforge.print(gdbforge.lua_dir())

gdbforge.clear()
  Clear the bound Lua pane cell grid (game panes).
  Example: gdbforge.clear()

gdbforge.gdb_query(cmd [, timeout_sec])
  Run a debugger command and return its output: out, err (err is "" on success). Timeout default 120 s.
  Example: local out, err = gdbforge.gdb_query("info threads")

gdbforge.gdb_command(command)
  Run a raw GDB/MI or Delve command and return the raw reply. [Remote]
  Example: gdbforge.gdb_command("info frame")

gdbforge.gdb_ctrl_c()
  Interrupt the running program (^C).
  Example: gdbforge.gdb_ctrl_c()

gdbforge.run_control(action [, wait_sec])
  continue|next|step|finish|stepi|nexti|run|start|interrupt; waits up to wait_sec (default 5, max 60)
  for a stop. Returns {state, file, line}; state "running" = not stopped yet, no file = exited. [Remote]
  Example: local s = gdbforge.run_control("next"); gdbforge.print(s.file, s.line)

gdbforge.evaluate(expression)
  Value of an expression in the current frame, as text (program must be stopped). [Remote]
  Example: gdbforge.print(gdbforge.evaluate("x"))

gdbforge.read_memory(address, length)
  Read 1-4096 bytes of target memory → {address, length, hex}. GDB only. [Remote]
  Example: gdbforge.print(gdbforge.read_memory("0x20000000", 16).hex)

gdbforge.registers([name…])
  Registers as {name = hex}. No names = general registers; "all" = every register. GDB only. [Remote]
  Example: local r = gdbforge.registers("pc", "sp")

gdbforge.list_breakpoints() / gdbforge.list_threads() / gdbforge.list_frames()
  Breakpoints, threads or call-stack frames from the shared model (same as the panes). [Remote]
  Example: gdbforge.print(gdbforge.list_frames())

gdbforge.set_breakpoint(file, line) / gdbforge.clear_breakpoint(file, line)
  Set or clear a breakpoint at file:line, like Space in the Code pane. [Remote]
  Example: gdbforge.set_breakpoint("main.c", 42)

gdbforge.set_kgdb_mode([on])
  Switch kgdb mode on (default) or off.
  Example: gdbforge.set_kgdb_mode(true)

gdbforge.uart_send(device, [baud,] line)
  Write a line (CRLF appended) to a serial device; baud default 115200.
  Example: gdbforge.uart_send("/dev/ttyUSB0", "reboot")

gdbforge.open_serial_terminal(device [, baud]) / gdbforge.close_serial_terminal()
  Open or close the shared serial terminal (baud default 115200).
  Example: gdbforge.open_serial_terminal("/dev/ttyUSB0", 115200)

gdbforge.serial_send(line) / gdbforge.serial_owner() / gdbforge.serial_switch_owner(mode)
  Send a line on the shared serial line, read or change its owner.
  Example: gdbforge.print(gdbforge.serial_owner())

gdbforge.serial_switch_gdb() / gdbforge.serial_switch_console() / gdbforge.begin_debug_entry()
  Hand the serial line to the debugger (kgdb mode on) or back to the console; start the debug entry sequence.
  Example: gdbforge.serial_switch_gdb()

gdbforge.serial_sysrq_delayed([delay_sec])
  Send SysRq-g after a delay (default 2 s).
  Example: gdbforge.serial_sysrq_delayed(1)

gdbforge.serial_debugger_pty() / gdbforge.serial_terminal_pty()
  Pty paths of the debugger side and the terminal side of the serial line.
  Example: gdbforge.print(gdbforge.serial_debugger_pty())

gdbforge.spawn_serial_console(pty [, baud])
  Run a serial console on a pty (baud default 115200).
  Example: gdbforge.spawn_serial_console(gdbforge.serial_terminal_pty())

Services over REST / MCP
  Functions marked [Remote] are also served by gdbforge -serve 127.0.0.1:PORT (loopback only).
  A bearer token is required: -serve-token T or GDBFORGE_TOKEN; clients send "Authorization: Bearer T".
  GET /api lists them, POST /api/<name> with a JSON body calls one, /mcp is the MCP endpoint.
  Example: curl -H "Authorization: Bearer $GDBFORGE_TOKEN" -XPOST localhost:7777/api/evaluate -d '{"expression":"x"}'

pane.set_cell(x, y, ch [, color])
  Draw one cell in a ModeLua pane (:lua snake, etc.).
  Example: pane.set_cell(0, 0, "#", "red")

pane.clear()
  Clear pane cells only (not :b io scrollback).
  Example: pane.clear()

pane.size()
  Returns width, height of the Lua pane.
  Example: local w, h = pane.size()

Script help
  Per-workflow usage: :lua <name> help  (calls that script's help() if defined).
  Example: :lua remotegdb help
